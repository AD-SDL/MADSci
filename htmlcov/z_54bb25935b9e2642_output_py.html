61def determine_output_format(ctx: click.Context) -> OutputFormat: 

62 """Determine the output format from Click context flags. 

63 

64 Precedence: ``--json`` > ``--yaml`` > ``-q``/``--quiet`` > table (default). 

65 

66 Args: 

67 ctx: Click context populated by the top-level ``madsci`` group. 

68 

69 Returns: 

70 The :class:`OutputFormat` to use. 

71 """ 

72 obj = ctx.obj or {} 

73 if obj.get("json"): 

74 return OutputFormat.JSON 

75 if obj.get("yaml"): 

76 return OutputFormat.YAML 

77 if obj.get("quiet"): 

78 return OutputFormat.QUIET 

79 return OutputFormat.TABLE

87def get_console(ctx: click.Context) -> Console: 

88 """Get Rich Console from Click context, with fallback. 

89

99 if ctx.obj and "console" in ctx.obj: 

100 return ctx.obj["console"]

104def _serialize(data: Any) -> Any: 

105 """Convert *data* to a JSON-safe Python structure. 

106 

107 Handles Pydantic models, lists of models, and plain dicts/values. 

108 """ 

109 if isinstance(data, BaseModel): 

110 return data.model_dump(mode="json") 

111 if isinstance(data, list): 

112 return [_serialize(item) for item in data] 

113 return data

116def _output_quiet(console: Console, data: Any, quiet_key: str | None) -> None: 

117 """Render data in quiet mode (minimal output).""" 

118 serialised = _serialize(data) 

119 if quiet_key and isinstance(serialised, dict): 

120 console.print(str(serialised.get(quiet_key, ""))) 

121 elif isinstance(serialised, list) and quiet_key: 

122 for item in serialised: 

123 val = item.get(quiet_key, item) if isinstance(item, dict) else item 

124 console.print(str(val)) 

125 else: 

126 console.print(str(serialised))

129def _output_table( 

130 console: Console, 

131 data: Any, 

132 title: str | None, 

133 columns: Sequence[ColumnDef] | None, 

134) -> None: 

135 """Render data as a Rich table or fallback format.""" 

136 # List with column definitions -> proper table 

137 if columns and isinstance(data, (list, tuple)) and data: 

138 _print_column_table(console, data, title, columns) 

139 return 

140 

141 # Single Pydantic model or dict -> key/value table 

142 mapping = data.model_dump(mode="json") if isinstance(data, BaseModel) else data 

143 if isinstance(mapping, dict): 

144 _print_kv_table(console, mapping, title) 

145 return 

146 

147 # Plain list (no column defs) -> bullet list 

148 if isinstance(data, list): 

149 for item in data: 

150 console.print(f" \u2022 {item}") 

151 return 

152 

153 # Scalar fallback 

154 console.print(data)

157def _print_column_table( 

158 console: Console, 

159 data: Sequence[Any], 

160 title: str | None, 

161 columns: Sequence[ColumnDef], 

162) -> None: 

163 """Render a list of items as a table with column definitions.""" 

164 table = Table(title=title) 

165 for col in columns: 

166 table.add_column(col.header, style=col.style or "", max_width=col.max_width) 

167 

168 for item in data: 

169 row_dict = _serialize(item) if isinstance(item, BaseModel) else item 

170 if isinstance(row_dict, dict): 

171 table.add_row(*(str(row_dict.get(col.key, "")) for col in columns)) 

172 else: 

173 table.add_row(str(row_dict)) 

174 console.print(table)

177def _print_kv_table( 

178 console: Console, 

179 mapping: dict[str, Any], 

180 title: str | None, 

181) -> None: 

182 """Render a dict as a two-column key/value table.""" 

183 table = Table(title=title, show_header=False) 

184 table.add_column("Key", style="cyan") 

185 table.add_column("Value") 

186 for key, value in mapping.items(): 

187 table.add_row(str(key), str(value)) 

188 console.print(table)

191def output_result( 

192 console: Console, 

193 data: Any, 

194 format: str = "text", 

195 title: str | None = None, 

196 columns: Sequence[ColumnDef] | None = None, 

197 quiet_key: str | None = None, 

198) -> None: 

199 """Output data in the requested format. 

200 

201 Enhanced to support: 

202 * Pydantic model auto-serialisation 

203 * :class:`ColumnDef`-based table rendering 

204 * ``quiet`` mode (prints only the *quiet_key* value) 

205 * Lists of Pydantic models 

206 

207 Args: 

208 console: Rich console instance. 

209 data: Data to output (dict, list, Pydantic model, or scalar). 

210 format: Output format (``"text"``, ``"json"``, ``"yaml"``, ``"quiet"``). 

211 title: Optional title for formatted output. 

212 columns: Optional column definitions for table rendering. 

213 quiet_key: Key whose value is printed in ``quiet`` mode. 

214 """ 

215 if format == "quiet": 

216 _output_quiet(console, data, quiet_key) 

217 elif format == "json": 

218 console.print_json(json.dumps(_serialize(data), default=str)) 

219 elif format == "yaml": 

220 console.print(yaml.dump(_serialize(data), default_flow_style=False)) 

221 else: 

222 _output_table(console, data, title, columns)
