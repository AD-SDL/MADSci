29class DetailPanel(Static): 

30 """Key-value detail display with titled sections. 

31 

32 Provides a standard pattern for detail panels throughout the TUI: 

33 a title, followed by one or more sections each containing indented 

34 key-value pairs. 

35 

36 Usage:: 

37 

38 panel = DetailPanel() 

39 panel.update_content( 

40 title="Node: robot_arm", 

41 sections=[ 

42 DetailSection("General", { 

43 "URL": "http://localhost:2000/", 

44 "Status": "[green]connected[/green]", 

45 }), 

46 DetailSection("Actions", { 

47 "pick": "Pick up an item", 

48 "place": "Place an item", 

49 }), 

50 ], 

51 ) 

52 panel.clear_content() 

53 """ 

54 

55 DEFAULT_CSS = """ 

56 DetailPanel { 

57 height: auto; 

58 min-height: 4; 

59 } 

60 """

77 def compose(self) -> ComposeResult: 

78 """Compose the panel with a content label.""" 

79 yield Label( 

80 f"[dim]{self._placeholder}[/dim]", 

81 id="detail-panel-content", 

82 ) 

83 

84 def update_content( 

85 self, 

86 title: str, 

87 sections: list[DetailSection], 

88 ) -> None: 

89 """Update the panel with new content. 

90 

91 Args: 

92 title: Main title displayed at the top of the panel. 

93 sections: List of sections to display below the title. 

94 """ 

95 lines: list[str] = [f"[bold]{title}[/bold]"] 

96 

97 for section in sections: 

98 lines.append("") 

99 lines.append(f" [bold]{section.title}[/bold]") 

100 if section.fields: 

101 # Calculate max key length for alignment 

102 max_key_len = max(len(k) for k in section.fields) 

103 for key, value in section.fields.items(): 

104 lines.append(f" {key:<{max_key_len}} {value}") 

105 else: 

106 lines.append(" [dim]No data[/dim]") 

107 

108 content = self.query_one("#detail-panel-content", Label) 

109 content.update("\n".join(lines)) 

110 

111 def clear_content(self) -> None: 

112 """Reset the panel to its placeholder state.""" 

113 content = self.query_one("#detail-panel-content", Label) 

114 content.update(f"[dim]{self._placeholder}[/dim]")
