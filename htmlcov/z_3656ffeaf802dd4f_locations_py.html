33def _matches_filter( 

34 location: Location, 

35 search: str, 

36) -> bool: 

37 """Check whether a location matches the current search text. 

38 

39 Args: 

40 location: Location model instance. 

41 search: Search text (matched against location_name, case-insensitive). 

42 

43 Returns: 

44 True if the location passes the filter. 

45 """ 

46 if search: 

47 name = location.location_name 

48 if search.lower() not in name.lower(): 

49 return False 

50 return True

53def _build_general_section(location: Location) -> DetailSection: 

54 """Build the general info section for the detail panel. 

55 

56 Args: 

57 location: Location model instance. 

58 

59 Returns: 

60 DetailSection with general fields. 

61 """ 

62 managed_by_str = location.managed_by.value.upper() 

63 

64 owner = getattr(location, "owner", None) 

65 if owner and getattr(owner, "node_id", None): 

66 owner_str = str(owner.node_id) 

67 else: 

68 owner_str = "N/A" 

69 

70 fields: dict[str, str] = { 

71 "Name": location.location_name or "Unknown", 

72 "ID": location.location_id, 

73 "Managed By": managed_by_str, 

74 "Owner": owner_str, 

75 "Allow Transfers": str(location.allow_transfers), 

76 } 

77 if location.location_template_name: 

78 fields["Template"] = str(location.location_template_name) 

79 if location.description: 

80 fields["Description"] = str(location.description)[:100] 

81 return DetailSection("General", fields)

101def _build_representations_section(location: Location) -> DetailSection | None: 

102 """Build the representations section for the detail panel. 

103 

104 Args: 

105 location: Location model instance. 

106 

107 Returns: 

108 DetailSection if representations exist, else None. 

109 """ 

110 representations = location.representations or {}

135def _build_reservation_section(location: Location) -> DetailSection | None: 

136 """Build the reservation section for the detail panel. 

137 

138 Args: 

139 location: Location model instance. 

140 

141 Returns: 

142 DetailSection if a reservation exists, else None. 

143 """ 

144 reservation = location.reservation 

145 if not reservation: 

146 return None 

147 fields: dict[str, str] = {} 

148 owned_by = reservation.owned_by 

149 if owned_by: 

150 for key in ("user_id", "experiment_id", "campaign_id"): 

151 value = getattr(owned_by, key, None) 

152 if value: 

153 fields[key.replace("_", " ").title()] = str(value) 

154 created = reservation.created 

155 if created: 

156 fields["Created"] = format_timestamp(created, short=True) 

157 expires = reservation.expires 

158 if expires: 

159 fields["Expires"] = format_timestamp(expires, short=True) 

160 if fields: 

161 return DetailSection("Reservation", fields) 

162 return None

238 async def refresh_data(self) -> None: 

239 """Refresh location data from the location manager.""" 

240 self.locations_data.clear() 

241 self._location_ids.clear() 

242 

243 try: 

244 client = self._get_location_client() 

245 locations = await client.async_get_locations() 

246 for location in locations: 

247 self.locations_data[location.location_id] = location 

248 self._location_ids.append(location.location_id) 

249 except Exception: 

250 self.notify("Failed to reach Location Manager", timeout=3) 

251 

252 self._populate_table()

254 def _populate_table(self) -> None: 

255 """Populate the locations table with filtered data.""" 

256 table = self.query_one("#locations-table", DataTable) 

257 

258 filtered = [ 

259 loc_id 

260 for loc_id in self._location_ids 

261 if loc_id in self.locations_data 

262 and _matches_filter( 

263 self.locations_data[loc_id], 

264 self._current_search, 

265 ) 

266 ] 

267 

268 with preserve_cursor(table): 

269 table.clear() 

270 

271 for loc_id in filtered: 

272 loc = self.locations_data[loc_id] 

273 name = loc.location_name or "Unknown" 

274 managed_by_str = loc.managed_by.value.upper() 

275 loc_id_str = loc_id or "-" 

276 template = loc.location_template_name or "-" 

277 resource = loc.resource_id 

278 resource_display = resource or "None" 

279 transfers = "Yes" if loc.allow_transfers else "No" 

280 reservation = "Reserved" if loc.reservation else "-" 

281 table.add_row( 

282 name, 

283 managed_by_str, 

284 loc_id_str, 

285 template, 

286 resource_display, 

287 transfers, 

288 reservation, 

289 ) 

290 

291 if not filtered: 

292 table.add_row( 

293 "[dim]No locations found[/dim]", 

294 "-", 

295 "-", 

296 "-", 

297 "-", 

298 "-", 

299 "-", 

300 )

311 def on_data_table_row_selected(self, event: DataTable.RowSelected) -> None: 

312 """Handle row selection in the locations table. 

313 

314 Args: 

315 event: The row selected event. 

316 """ 

317 table = event.data_table 

318 row_key = event.row_key 

319 if not row_key: 

320 return 

321 

322 row = table.get_row(row_key) 

323 row_id = str(row[2]) 

324 

325 # Find location by matching full ID 

326 if row_id in self.locations_data: 

327 self.selected_location_id = row_id 

328 self._update_detail_panel(self.locations_data[row_id])

330 def _update_detail_panel(self, location: Location) -> None: 

331 """Update the detail panel with location data. 

332 

333 Args: 

334 location: Location model instance. 

335 """ 

336 detail_panel = self.query_one("#location-detail-panel", DetailPanel) 

337 name = location.location_name or "Unknown" 

338 

339 sections: list[DetailSection] = [_build_general_section(location)] 

340 sections.append(_build_resource_section(location)) 

341 

342 representations_section = _build_representations_section(location) 

343 if representations_section: 

344 sections.append(representations_section) 

345 

346 bindings_section = _build_node_bindings_section(location) 

347 if bindings_section: 

348 sections.append(bindings_section) 

349 

350 reservation_section = _build_reservation_section(location) 

351 if reservation_section: 

352 sections.append(reservation_section) 

353 

354 detail_panel.update_content(title=name, sections=sections)
